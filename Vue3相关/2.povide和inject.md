# provide/inject 的本质是：利用原型链实现跨层级数据共享，每个组件实例上挂一个 provides 对象，inject 沿原型链向上查找

provide/inject 的本质是：利用原型链实现跨层级数据共享，每个组件实例上挂一个 provides 对象，inject 沿原型链向上查找

面试回答： 组件createcomponentinstance实例的时候就有provide 从父组件身上找，如果是根组件直接继承app.context.provides. 然后provide的时候，如果父和子provides相同，通过object.create()创建组件独立的provides对象，但是还能通过原型链访问父组件的所有数据

```js
export function provide<T>(key: InjectionKey<T> | string | number, value: T) {
  // 1. 必须在组件 setup 里调用
  if (!currentInstance) {
    if (__DEV__) {
      warn(`provide() can only be used inside setup().`)
    }
  } else {
    // 2. 拿到当前组件实例的 provides
    let provides = currentInstance.provides

    // 3. ★★★ 核心逻辑：原型链隔离
    const parentProvides =
      currentInstance.parent && currentInstance.parent.provides

// 第一次进入
    if (parentProvides === provides) {
      // 当前 provides 还是指向父组件的同一个对象引用
      // 如果直接写入，会污染父组件的 provides！
      // 所以：以父 provides 为原型，创建新对象（原型链继承）
    //   # 写入的 key 只在自己这层，不会污染父组件
// # 通过原型链，仍然能访问到祖先组件提供的所有数据
      provides = currentInstance.provides = Object.create(parentProvides)
      //                                     ↑ 关键！
      // 现在：当前组件有了自己独立的 provides 对象
      // 但通过原型链，仍然能访问到父组件提供的所有数据
    }

    // 4. 把 key/value 写入当前组件自己的 provides
    provides[key as string] = value
  }
}
```

```js
export function inject(
  key: InjectionKey<any> | string,
  defaultValue?: unknown,
  treatDefaultAsFactory = false  // 默认值是否作为工厂函数调用
) {
  // 1. 拿当前组件实例（或者 getCurrentInstance()）
  const instance = currentInstance || currentRenderingInstance

  if (instance) {
    // 2. ★ 从父组件的 provides 开始找（不是自己的！）
    //    防止组件 inject 到自己 provide 的值
    const provides =
      instance.parent == null
        ? instance.appContext.provides   // 根组件：从 app 级 provides 找
        : instance.parent.provides       // 普通组件：从父组件的 provides 找

    if (provides && (key as string | symbol) in provides) {
      // 3. in 操作符会沿原型链查找 ★
      //    这里利用了 provide 时构建的原型链！
      return provides[key as string]
    } else if (arguments.length > 1) {
      // 4. 没找到，使用默认值
      return treatDefaultAsFactory && isFunction(defaultValue)
        ? defaultValue.call(instance.proxy)  // 默认值是函数 → 调用它
        : defaultValue                        // 直接返回默认值
    } else if (__DEV__) {
      warn(`injection "${String(key)}" not found.`)
    }
  }
}
```
