// Vue3：v-if 优先级 > v-for
// 编译结果：
// isShow ? list.map(item => <div>item</div>) : null

# // 但问题：v-if 里用了 v-for 的 item → item 找不到！

// ❌ 错误写法：

<div v-for="item in list" v-if="item.isShow">
  // Vue3 中 v-if 先执行，item 还不存在！
</div>

// ✅ 正确写法一：template 包裹
<template v-for="item in list">

  <div v-if="item.isShow">{{ item.name }}</div>
</template>

// ✅ 正确写法二：computed 过滤
const filteredList = computed(() =>
list.value.filter(item => item.isShow)
)
// 模板里直接遍历 filteredList，不需要 v-if
