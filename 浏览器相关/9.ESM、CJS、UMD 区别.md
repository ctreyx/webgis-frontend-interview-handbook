┌──────────────────┬──────────────────────────────────────┐
│  格式             │  说明                                │
├──────────────────┼──────────────────────────────────────┤
│  ESM              │  import / export                     │
│  (ES Module)      │  现代浏览器和打包工具用              │
│                   │  支持 Tree Shaking                   │
│                   │  文件后缀一般 .mjs 或 .js            │
├──────────────────┼──────────────────────────────────────┤
│  CJS              │  require / module.exports            │
│  (CommonJS)       │  Node.js 用                          │
│                   │  不支持 Tree Shaking                 │
│                   │  文件后缀一般 .cjs 或 .js            │
├──────────────────┼──────────────────────────────────────┤
│  UMD              │  兼容 ESM + CJS + 全局变量           │
│  (Universal)      │  可以用 script 标签直接引入           │
│                   │  体积最大，兼容性最好                 │
│                   │  CDN 引入一般用这个                   │
└──────────────────┴──────────────────────────────────────┘


# package.json 关键字段
{
  "name": "@yourlib/components",
  "version": "1.0.0",
  
  "main": "./dist/index.cjs.js",
  "module": "./dist/index.esm.js",
  
  "exports": {
    ".": {
      "import": "./dist/index.esm.js",
      "require": "./dist/index.cjs.js"
    },
    "./style": "./dist/style.css"
  },

  "types": "./dist/index.d.ts",
  
  "files": ["dist"],
  
  "peerDependencies": {
    "vue": "^3.0.0"
  }
}

┌──────────────────┬──────────────────────────────────────┐
│  字段             │  作用                                │
├──────────────────┼──────────────────────────────────────┤
│  main            │  CJS 入口                            │
│                  │  require('yourlib') 时用这个         │
├──────────────────┼──────────────────────────────────────┤
│  module          │  ESM 入口                            │
│                  │  import 时优先用这个                  │
│                  │  打包工具会优先读 module              │
│                  │  因为 ESM 可以 Tree Shaking          │
├──────────────────┼──────────────────────────────────────┤
│  exports         │  Node.js 12+ 的新规范                │
│                  │  优先级最高，覆盖 main 和 module     │
│                  │  可以精确控制不同引入方式的入口        │
├──────────────────┼──────────────────────────────────────┤
│  types           │  TypeScript 类型声明入口              │
├──────────────────┼──────────────────────────────────────┤
│  files           │  npm 发布时包含哪些文件               │
│                  │  只发 dist，不发源码                   │
├──────────────────┼──────────────────────────────────────┤
│  peerDependencies│  宿主项目必须安装的依赖               │
│                  │  Vue 不打进包里，用宿主的              │
│                  │  避免两份 Vue 冲突                    │
└──────────────────┴──────────────────────────────────────┘
