# 2. Pinia 底层是怎么实现响应式的？defineStore 内部做了什么？

// defineStore 内部简化原理

```js
function defineStore(idOrOptions, setup) {
  // 1. 解析参数
  const id = typeof idOrOptions === "string" ? idOrOptions : idOrOptions.id;

  // 2. 返回 useStore 函数
  function useStore() {
    // 3. 从当前 app 实例获取 pinia
    const pinia = inject(piniaSymbol); // 你说到了 ✅

    // 4. 如果 store 已存在，直接返回（单例）
    if (pinia._s.has(id)) {
      return pinia._s.get(id);
    }

    // 5. 执行 setup 函数，用 Vue3 响应式包裹
    // 这里是关键！
    const setupStore = setup();

    // 6. 把 setup 返回的所有 ref/reactive 收集起来
    // 这就是为什么 Pinia 的响应式底层就是 Vue3 的 ref/reactive
    const store = reactive(setupStore);

    // 7. 注册到 pinia._s（Map 结构）
    pinia._s.set(id, store);

    return store;
  }

  return useStore;
}

// 所以你在 setup store 里写的：
export const useMapStore = defineStore("map", () => {
  const zoom = ref(10); // 就是普通的 Vue3 ref
  const center = ref([116, 39]); // 完全没有魔法

  return { zoom, center };
});
// Pinia 只是帮你把这些收集起来，注册到全局
```
