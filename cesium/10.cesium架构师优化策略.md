## 真实可用版 — 只做两件事：检测 + 调参

### 完整代码（一个文件搞定）

```javascript
import * as Cesium from 'cesium'

/**
 * Cesium 性能自适应
 * 思路：静态信号初筛 → FPS实测定档 → 运行时动态降档
 */

// ─── 档位参数 ───────────────────────────────────────────
const TIERS = {
  HIGH: {
    maximumScreenSpaceError: 16,   // 3D Tiles LOD阈值，越大越省性能  SSE 越大：允许更大屏幕误差 → 更少细化、更少瓦片、更高性能，画质变糙
    resolutionScale: 1.0,          // 渲染分辨率
    requestRenderMode: false,      // 连续渲染  如果你的政府内网业务是“浏览 + 点击查询”为主、镜头不一直动，那么开启按需渲染能显著省电省 CPU（静止时不渲染）。
    msaaSamples: 4,               // 抗锯齿
    fog: true,
  },
  MEDIUM: {
    maximumScreenSpaceError: 32,
    resolutionScale: 0.85,
    requestRenderMode: true,       // 静止不渲染，省CPU
    msaaSamples: 2,
    fog: true,
  },
  LOW: {
    maximumScreenSpaceError: 64,
    resolutionScale: 0.65,
    requestRenderMode: true,
    msaaSamples: 1,
    fog: false,
  },
}

// ─── 检测设备性能，返回档位 ──────────────────────────────
async function detectTier() {
  let score = 0

  // 1. CPU核心数（兼容性好，基本所有浏览器可用）
  const cores = navigator.hardwareConcurrency || 4
  if (cores >= 8) score += 40
  else if (cores >= 4) score += 20
  else score += 5

  // 2. 内存（deviceMemory部分浏览器不支持，有就用）
  const memory = navigator.deviceMemory  // 单位GB，值为0.25/0.5/1/2/4/8之一
  if (memory != null) {
    if (memory >= 8) score += 30
    else if (memory >= 4) score += 15
    else score += 0
  } else {
    score += 10  // 不支持就给个中间分
  }

  // 3. WebGL能力（反映GPU档次）
  const canvas = document.createElement('canvas')
  const gl = canvas.getContext('webgl2') || canvas.getContext('webgl')
  if (gl) {
    // 最大纹理尺寸：高端独显16384，集显/低端4096
    const maxTex = gl.getParameter(gl.MAX_TEXTURE_SIZE)
    if (maxTex >= 16384) score += 30
    else if (maxTex >= 8192) score += 15
    else score += 5
  }

  // 4. FPS实测（最真实的信号，跑2秒rAF）
  const fps = await measureFPS(2000)
  if (fps >= 55) score += 30
  else if (fps >= 30) score += 15
  else score += 0

  console.log(`[性能检测] cores:${cores} memory:${memory} fps:${fps.toFixed(1)} score:${score}`)

  if (score >= 90) return 'HIGH'
  if (score >= 50) return 'MEDIUM'
  return 'LOW'
}

// ─── FPS测量 ─────────────────────────────────────────────
function measureFPS(duration = 2000) {
  return new Promise(resolve => {
    let count = 0
    const start = performance.now()

    const tick = () => {
      count++
      if (performance.now() - start < duration) {
        requestAnimationFrame(tick)
      } else {
        resolve(count / (duration / 1000))
      }
    }
    requestAnimationFrame(tick)
  })
}

// ─── 应用档位参数到Cesium ────────────────────────────────
function applyTier(viewer, tilesets, tier) {
  const config = TIERS[tier]
  const scene = viewer.scene

  // 渲染分辨率（降低此值对性能提升最明显）
  scene.pixelRatio = config.resolutionScale

  // 按需渲染（中低档：静止时不重绘，CPU几乎归零）
  scene.requestRenderMode = config.requestRenderMode

  // 雾效
  scene.fog.enabled = config.fog

  // 抗锯齿（WebGL2才支持MSAA）
  if (scene.msaaSamples !== undefined) {
    scene.msaaSamples = config.msaaSamples
  }

  // 3D Tiles LOD（每个tileset都要设置）
  tilesets.forEach(tileset => {
    tileset.maximumScreenSpaceError = config.maximumScreenSpaceError

    // 动态SSE：让远处图层自动降精度，近处保持清晰
    tileset.dynamicScreenSpaceError = true
    tileset.dynamicScreenSpaceErrorDensity = 0.00278
    tileset.dynamicScreenSpaceErrorFactor = tier === 'LOW' ? 8.0 : 4.0
  })

  // 按需渲染模式下手动触发一次重绘，否则画面不更新
  if (config.requestRenderMode) {
    scene.requestRender()
  }

  console.log(`[Cesium] 应用档位: ${tier}`, config)
}

// ─── 运行时FPS监控，持续采样动态降档 ─────────────────────
function startFPSMonitor(viewer, tilesets, getCurrentTier, onTierChange) {
  const TIERS_ORDER = ['LOW', 'MEDIUM', 'HIGH']
  let lowFpsCount = 0    // 防抖：连续低FPS次数

  // 每10秒采样一次（间隔不要太短，避免采样误差）
  const intervalId = setInterval(async () => {
    const fps = await measureFPS(3000)

    if (fps < 25) {
      lowFpsCount++
      // 连续2次低FPS才降档，防止网络抖动误触发
      if (lowFpsCount >= 2) {
        lowFpsCount = 0
        const currentIdx = TIERS_ORDER.indexOf(getCurrentTier())
        if (currentIdx > 0) {
          const newTier = TIERS_ORDER[currentIdx - 1]
          onTierChange(newTier)
        }
      }
    } else {
      lowFpsCount = 0  // FPS正常则重置计数
    }

    console.log(`[FPS监控] ${fps.toFixed(1)} fps，当前档位: ${getCurrentTier()}`)
  }, 10000)

  // 返回清理函数
  return () => clearInterval(intervalId)
}

// ─── 主入口：对外暴露的接口 ──────────────────────────────
export async function initCesiumPerformance(viewer, tilesets = []) {
  let currentTier = 'MEDIUM'

  // 先应用中档，保证首屏不卡（检测需要2~3秒）
  applyTier(viewer, tilesets, 'MEDIUM')

  // 异步检测，不阻塞地图显示
  const detectedTier = await detectTier()
  if (detectedTier !== currentTier) {
    currentTier = detectedTier
    applyTier(viewer, tilesets, currentTier)
  }

  // 启动持续监控
  const stopMonitor = startFPSMonitor(
    viewer,
    tilesets,
    () => currentTier,
    (newTier) => {
      console.warn(`[性能自适应] 自动降档: ${currentTier} → ${newTier}`)
      currentTier = newTier
      applyTier(viewer, tilesets, newTier)
    }
  )

  return {
    getCurrentTier: () => currentTier,
    // 手动切换（给UI按钮用）
    setTier: (tier) => {
      currentTier = tier
      applyTier(viewer, tilesets, tier)
    },
    // 新加载的tileset自动继承当前档位
    addTileset: (tileset) => {
      tilesets.push(tileset)
      const config = TIERS[currentTier]
      tileset.maximumScreenSpaceError = config.maximumScreenSpaceError
      tileset.dynamicScreenSpaceError = true
    },
    destroy: stopMonitor,
  }
}
```

### 使用（3行接入）

```javascript
const viewer = new Cesium.Viewer('cesiumContainer')
const tileset = await Cesium.Cesium3DTileset.fromUrl('/3dtiles/city/tileset.json')
viewer.scene.primitives.add(tileset)

// 接入性能适配，传viewer和所有tileset
const perf = await initCesiumPerformance(viewer, [tileset])

// 后续动态加载的图层
const tileset2 = await Cesium.Cesium3DTileset.fromUrl('/3dtiles/pipeline/tileset.json')
viewer.scene.primitives.add(tileset2)
perf.addTileset(tileset2)  // 自动继承当前档位

// UI手动切换按钮
document.getElementById('btn-low').onclick = () => perf.setTier('LOW')
```

## 面试怎么说（真实版，90秒）

> **"我们这个政府内网项目，终端环境差异很大，有人用工作站，有人用普通办公机，Cesium 默认参数在低端机上会直接卡。**
>
> **我做了一套性能自适应，分三步：**
>
> **第一步，启动时检测。** 用 `hardwareConcurrency` 读核心数，`deviceMemory` 读内存，再用 WebGL 读最大纹理尺寸来判断 GPU 档次，这些信号加权出一个初始分。然后跑 2 秒 `requestAnimationFrame` 测实际帧率，实测 FPS 是最准的信号。综合下来分出高中低三档。
>
> **第二步，按档位调参。** 核心就三个参数：`maximumScreenSpaceError` 控制 3D Tiles 的 LOD 阈值，低档设 64 高档设 16，直接决定同屏渲染多少瓦片；`pixelRatio` 控制渲染分辨率，低档 0.65 对填充率压力下降非常明显；`requestRenderMode` 在中低档开启，静止时完全不重绘，CPU 占用几乎归零。
>
> **第三步，运行时监控。** 每 10 秒采样一次 FPS，连续 2 次低于 25 才触发降档，防止单次网络抖动误判。
>
> **首屏处理上有个细节**：检测本身要 2~3 秒，所以进来先用中档渲染，检测完再切，不阻塞用户看到地图。"

## 面试追问答案

| 追问 | 真实回答 |
|---|---|
| **deviceMemory 不准怎么办？** | 它本来就是模糊值，故意设计成这样防止指纹追踪，所以我只给它 30% 权重，FPS 实测是最重的 |
| **requestRenderMode 有什么坑？** | 开了之后数据更新不会自动触发重绘，我们所有修改 Entity 属性的地方统一封装了一层，末尾手动调 `scene.requestRender()` |
| **maximumScreenSpaceError 怎么确定这三个值？** | 我们先在低端机上用默认值16跑，发现 FPS 稳定在 8~10，然后逐步调大，到 64 的时候 FPS 上来了，业务也能接受这个精度，就定下来了 |
| **为什么不用WebGL渲染器信息判断显卡？** | `WEBGL_debug_renderer_info` 这个扩展 Firefox 已经禁用，Chrome 也在限制，不能依赖。而且就算拿到字符串也没法可靠地映射到性能档位 |
